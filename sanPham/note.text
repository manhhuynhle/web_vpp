create database HOC_TOT
USE HOC_TOT

CREATE TABLE san_pham (
    ma_san_pham INT AUTO_INCREMENT PRIMARY KEY,
    ten_san_pham VARCHAR(255) NOT NULL,
    slug VARCHAR(255) UNIQUE,
    ma_danh_muc INT,
    thuong_hieu VARCHAR(100) DEFAULT 'Khác',
    ma_san_xuat VARCHAR(50),
    mo_ta TEXT,
    gia_goc DECIMAL(12,2) NOT NULL DEFAULT 0,
    gia_ban DECIMAL(12,2) NOT NULL DEFAULT 0,
    so_luong_ton INT NOT NULL DEFAULT 0,
    hinh_anh VARCHAR(255),
    trang_thai TINYINT NOT NULL DEFAULT 1,
    noi_bat TINYINT NOT NULL DEFAULT 0,
    ngay_tao DATETIME DEFAULT CURRENT_TIMESTAMP,
    ngay_cap_nhat DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP
);
